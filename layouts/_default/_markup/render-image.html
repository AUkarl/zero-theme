{{- /* ============================================================ render-image.html — 正文 markdown 图片的渲染钩子（R44） ------------------------------------------------------------ 在这之前，正文里的 `![]()` 是 Hugo 原样输出的 ``： 没有 srcset、没有宽高、手机也只能下原图 —— 全站图片慢的主要来源。 这里接上「上传期三档变体」的清单： · 有变体：显示 720（备 360），`data-orig` 给灯箱 1080，带 width/height 防抖动 · 没变体（还没回填 / 第三方图源）：原样输出，只补 lazy 与解码提示 首图（.Ordinal == 0）给 eager + fetchpriority=high：它往往就是 LCP 元素， 以前它也是 lazy，等于让浏览器等布局完成才去下。 ============================================================ */ -}} {{- $src := .Destination -}} {{- $eager := eq .Ordinal 0 -}} {{- $alt := .PlainText -}} {{- if not $alt }}{{ $alt = .Text | plainify }}{{ end -}} {{- $v := partial "variants-lookup.html" (dict "src" $src "role" "article") -}} {{- if $v.ok -}} {{- else -}} {{- /* 没进清单的图：本地图在构建期取真尺寸补 width/height（CSS 是 max-width:100% + height:auto， 补上属性只是让浏览器提前留出这块比例，**不改任何视觉效果**）； 远程图与 svg 拿不到尺寸就照旧不写，绝不为了补属性在构建期去联网。 */ -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- $remote := or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "//") (hasPrefix $src "data:") -}} {{- $clean := strings.TrimPrefix "/" $src -}} {{- $noSize := or (hasSuffix (lower $clean) ".svg") (hasSuffix (lower $src) ".svg") -}} {{- if and (not $remote) (not $noSize) -}} {{- $res := $.Page.Resources.GetMatch $clean -}} {{- if and $res $res.Width $res.Height -}} {{- $w = $res.Width -}}{{- $h = $res.Height -}} {{- else -}} {{- $fp := printf "static/%s" $clean -}} {{- if fileExists $fp -}} {{- $ic := imageConfig $fp -}}{{- $w = $ic.Width -}}{{- $h = $ic.Height -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
