{{- /* ============================================================ page-title.html — 页面标题的唯一口径 · （head.html）、og:title / twitter:title / JSON-LD headline（seo.html）都取这一个值， 免得两处各自 `.Title | default 站名` 手抄两份、改一处就漂。 · 没写 title 的内容文件（后台发时刻只写 date/images）以前会一路空到 标签页只剩一个「· 站名」，分享卡片顶着的还是站点名。现在的退路是 .Title → 正文摘要（24 字）→ 日期 → 站名。 · 摘要只在真的没有 title 时才算：.Plain 要过一遍正文，不该每页都白算。 · 两个坑（本轮在台架上撞出来的）： ① 别拿 `.Section` 当兜底 —— 它是页面对象，打印出来是 "moments" 这种路径串； ② 别加「必须是 string」的类型护栏 —— `truncate` 返回的是 template.HTML， 那种护栏会把自己的成果抹成空串，等于没兜底。要挡的只是切片/映射这两种怪形状。 ============================================================ */ -}} {{- $siteCfg := partial "site-config.html" . -}} {{- $fallback := $siteCfg.name | default $siteCfg.navName | default "站点" -}} {{- $t := .Title -}} {{- /* Hugo 自动生成的分类法一览页（/author/、/categories/、/tags/）没有内容文件，.Title 会回退成 taxonomy 的英文名（Author / Categories / Tags），中文站就顶着一串英文当 <title>、og:title 与 h1。 这里按 i18n 换成语言化名字。三条边界： · 只认 kind=taxonomy（一览页）；单个作者/分类/标签的详情页（kind=term）标题是人写的，不动； · 站点自己在 content/zh/<某分类法>/_index.md 里写过标题的不动 —— 只有「标题正好等于自动回退值」时才换； · i18n 里没这个键就退回原值，不会把标题换成空串。 */ -}} {{- $taxoNames := dict "author" "author" "category" "category" "tag" "tag" -}} {{- $sing := lower (string .Data.Singular) -}} {{- if and (eq .Kind "taxonomy") (isset $taxoNames $sing) -}} {{- $cur := trim (printf "%s" $t) " " -}} {{- $auto := slice $sing (title $sing) (lower (string .Data.Plural)) (title (string .Data.Plural)) -}} {{- if or (not $cur) (in $auto $cur) -}} {{- $t = i18n (index $taxoNames $sing) | default $cur -}} {{- end -}} {{- end -}} {{- if or (reflect.IsSlice $t) (reflect.IsMap $t) }}{{ $t = "" }}{{ end -}} {{- if not $t }}{{ $t = printf "%s" (.Plain | plainify | replaceRE `\s+` " " | strings.TrimSpace | truncate 24 "…") }}{{ end -}} {{- if not $t }}{{ $t = .Date.Format "2006-01-02" }}{{ end -}} {{- if not $t }}{{ $t = $fallback }}{{ end -}} {{- return $t -}}
