{{- /* ============================================================ zero/strings.html — 把 front matter 里的分类/标签统一成字符串切片 ------------------------------------------------------------ 为什么需要：YAML 里不写引号的 `1984` 是数字，jsonify 出来就是 1984， 前端 search/filter 里 `tag.toLowerCase()` 会当场抛 TypeError，把整块筛选打死 （站长站上《1984》阅读笔记就是这个：归档与首页的搜索一起失效）。 这里统一 printf "%v" 转成字符串，传进来的可以是切片，也可以是单个值。 ============================================================ */ -}} {{- $out := slice -}} {{- with . -}} {{- if reflect.IsSlice . -}} {{- range . -}} {{- $s := printf "%v" . -}} {{- if $s }}{{ $out = $out | append $s }}{{ end -}} {{- end -}} {{- else -}} {{- $s := printf "%v" . -}} {{- if $s }}{{ $out = $out | append $s }}{{ end -}} {{- end -}} {{- end -}} {{- return $out -}}
