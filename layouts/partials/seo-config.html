{{- /* ============================================================ seo-config.html — SEO 参数的统一取法 ------------------------------------------------------------ 优先级：站点 data/seo.json（后台「前端管理 → SEO」写的） > params.toml 的 [seo] 段。 为什么要有这一层：SEO 的消费方有四处（head.html 的两处、seo.html、robots.txt）， 每处各写一遍「先看 data 再看 params」迟早会漏掉一个；合成一次、大家都读同一个结果。 键名一律 snake_case：params 的键在 Hugo 里会被小写化，而 JSON 的键保持原样， 两种拼法混用会让同一个参数在合成表里出现两个键。原来那四个驼峰键 （robotsDisallow / robotsAllow / robotsCrawlDelay / robotsHost）已改成 robots_disallow / robots_allow / robots_crawl_delay / robots_host。 布尔值可以放心 merge：这里是整键覆盖，不是 `| default`， 后台写 false 就是 false，不会被 params 里的 true 盖回去。 ============================================================ */ -}} {{- $p := .Site.Params.seo | default dict -}} {{- $d := hugo.Data.seo | default dict -}} {{- /* data 里空着的键不要盖掉 params：后台表单没填的项留空落盘， 直接 merge 会把 params 里的默认分享图/描述清成空。 注意旧模板的驼峰键会派生出值为空的下划线键（ogImageDefault → og_image_default: ""）， 这里把空值剔掉，正好让 params 那份继续生效。 */ -}} {{- $clean := dict -}} {{- range $k, $v := $d -}} {{- $empty := or (eq $v nil) (eq (printf "%v" $v) "") (and (reflect.IsSlice $v) (eq (len $v) 0)) -}} {{- if not $empty -}}{{- $clean = merge $clean (dict $k $v) -}}{{- end -}} {{- end -}} {{- /* 两个都是整键覆盖：data 有哪个键（且不是空值）就用哪个，没有的仍从 params 走 */ -}} {{- return (merge (dict) $p $clean) -}}
