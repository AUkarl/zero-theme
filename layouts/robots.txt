{{- /* ============================================================
  robots.txt — 由 SEO 参数生成（后台「前端管理 → SEO」优先，其次 params 的 [seo]）
  需要 config.toml 里 enableRobotsTXT = true：Hugo 只有在开了这个开关时
  才会渲染本文件（没开就压根不产出 robots.txt，不是「用内置规则覆盖本文件」）。
  ============================================================ */ -}}
{{- $seo := partial "seo-config.html" . -}}
{{- /* robots_allow / robots_disallow 在 data/seo.json 里是手写得到的（data 优先），
       漏掉方括号写成字符串时，下面两处 range 会直接把整站构建打死（Hugo: range can't iterate over …），
       而 Pages 的规矩是「构建失败就不换部署」—— 表现是站点静默停止更新。所以非数组先包一层。 */ -}}
{{- $disallow := $seo.robots_disallow | default (slice "/account/" "/admin/") -}}
{{- if not (reflect.IsSlice $disallow) -}}{{- $disallow = (slice $disallow) -}}{{- end -}}
{{- $allow := $seo.robots_allow | default (slice) -}}
{{- if not (reflect.IsSlice $allow) -}}{{- $allow = (slice $allow) -}}{{- end -}}
{{- $clean := $seo.robots_crawl_delay | default 0 -}}
User-agent: *
{{- range $allow }}
Allow: {{ . }}
{{- end }}
{{- range $disallow }}
Disallow: {{ . }}
{{- end }}
{{- if and $clean (gt (int $clean) 0) }}
Crawl-delay: {{ $clean }}
{{- end }}
{{- if (ne $seo.enable_robots_sitemap false) }}
Sitemap: {{ "sitemap.xml" | absURL }}
{{ if hugo.IsMultilingual }}
{{- range .Site.Languages }}
{{- $prefix := "" -}}
{{- if ne $.Site.Language.Lang .Lang }}{{ $prefix = printf "/%s" .Lang }}{{ end }}
Sitemap: {{ printf "%s/sitemap.xml" $prefix | absURL }}
{{- end }}
{{- end }}
{{- end }}
{{- with $seo.robots_host }}
Host: {{ . }}
{{- end }}
